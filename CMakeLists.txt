# Sherclawk uses HelloHTTPS's staged Certainly build and HTTP implementation.
cmake_minimum_required(VERSION 3.12)
project(sherclawk C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CERTAINLY_DIR)
    set(CERTAINLY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Certainly")
endif()
add_subdirectory("${CERTAINLY_DIR}" certainly-src)
set(CHAT_CORE chat.c json.c text.c network.c ../hello-https/http.c)
add_application(Sherclawk main.c agent.c tools.c build_project.c jobs.c ${CHAT_CORE} hello.r build/art.r CREATOR ShCk)
add_application(SherclawkProbe tools/probe.c agent.c tools.c ${CHAT_CORE} hello.r)
add_application(SherclawkWriteCheck tools/write-check.c tools.c json.c text.c hello.r)
add_application(SherclawkSearchCheck tools/search-check.c tools.c json.c text.c hello.r)
add_application(SherclawkEditCheck tools/edit-check.c tools.c json.c text.c hello.r)
add_application(SherclawkProjectCheck tools/project-check.c tools.c json.c text.c hello.r)
add_application(SherclawkBuildCheck tools/build-check.c build_project.c tools.c jobs.c json.c text.c hello.r)
add_application(SherclawkJobCheck tools/job-check.c jobs.c hello.r)
add_application(SherclawkScrollCheck tools/scroll-check.c agent.c tools.c build_project.c jobs.c ${CHAT_CORE} hello.r build/art.r CREATOR ShCk)
add_application(SherclawkHandoffCheck tools/handoff-check.c agent.c tools.c build_project.c jobs.c ${CHAT_CORE} hello.r build/art.r CREATOR ShCk)
foreach(app Sherclawk SherclawkProbe SherclawkScrollCheck SherclawkHandoffCheck SherclawkWriteCheck SherclawkEditCheck SherclawkSearchCheck SherclawkJobCheck SherclawkProjectCheck SherclawkBuildCheck)
    target_include_directories(${app} PRIVATE . ../hello-https)
    target_link_libraries(${app} certainly)
    target_compile_options(${app} PRIVATE -Wall -Wextra -ffunction-sections)
    set_target_properties(${app} PROPERTIES LINK_FLAGS "-Wl,-gc-sections")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config.local.h")
        target_compile_definitions(${app} PRIVATE SHERCLAWK_HAS_LOCAL_CONFIG=1)
    endif()
endforeach()
